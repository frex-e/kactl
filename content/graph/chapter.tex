	\subsection{Johnson's algorithm}
		All-pairs shortest paths with negative weights and no
		negative cycles. Run Bellman-Ford with every vertex at
		distance 0 (or from a virtual source); check for negative
		cycles. Reweight $w'(u,v)=w(u,v)+h(u)-h(v)$, then Dijkstra
		from each source.

	\subsection{Dinic variants and capacity edits}
		\texttt{Dinic} uses capacity scaling; \texttt{Dinic2}
		uses smaller edges and a blocking-flow DFS without scaling.
		Choose by workload: \texttt{Dinic2} is often faster,
		while scaling gives the stronger general worst-case bound.
		\texttt{Dinic2(n)} starts a fresh graph; to reset,
		assign \texttt{D = Dinic2(n)}. Save edge indices and
		original capacities yourself if needed.

		\noindent\textbf{Incremental flow (Dinic2).}
		Track the total separately: \texttt{F += D.calc(s,t)}.
		Between calls the residuals must stay nonnegative,
		each pair must sum to its current cap plus reverse cap,
		and flow must be conserved away from $s,t$.
		For edge $e=\texttt{adj[a][i]}$ to $b$, let
		$r=\texttt{adj[b][e.rev]}$ and $f=r.c-\text{rcap}$.
		Raise its cap by $d$: \texttt{e.c += d}, then augment.
		Lower its cap by $d$ without going below $f$:
		\texttt{e.c -= d}, then augment.

		\noindent\textbf{Lower below flow.}
		For new cap $c'<f$, put $x=f-c'$,
		\texttt{e.c=0; r.c-=x}.
		Reroute with \texttt{z=x-D.calc(a,b,x)};
		cancel the rest with \texttt{D.calc(a,s,z)} if $a\ne s$
		and \texttt{D.calc(t,b,z)} if $b\ne t$.
		Then \texttt{F -= z; F += D.calc(s,t)}.
		Self-loops carry zero flow and need no rerouting.
		Min-cut labels refer to the last exhausted $s$--$t$ call;
		a call stopped by a flow limit does not certify a cut.
	\subsection{Demands and lower bounds}
		\noindent\textbf{Lower bounds.}
		Edge $u\to v$ with capacity in $[L,R]$: force $L$
		units via \texttt{demand[v]+=L}, \texttt{demand[u]-=L},
		then replace the edge by capacity $R-L$.
		($L=0$ is an ordinary capacity-$R$ edge.)
		Original flow is $L$ plus flow on the rewritten
		cap-$(R-L)$ edge.

		\noindent\textbf{Circulation.}
		$d(v)$ is the forced net inflow (positive = excess).
		Residual flow must send net outflow $d(v)$ from each
		excess vertex. Add super-source $S$ and super-sink $T$:
		$S\to v$ of cap $d(v)$ if $d(v)>0$, $v\to T$ of cap
		$-d(v)$ if $d(v)<0$. Feasible iff $\sum d=0$ and max
		flow $S\to T$ equals $\sum_{d>0} d(v)$. A circulation
		is this check after the lower-bound rewrite.

		\noindent\textbf{Max $s$--$t$ flow.}
		Add $t\to s$ cap $\infty$, find a feasible
		circulation. Let $F$ be the flow on $t\to s$;
		delete $S$, $T$, and $t\to s$, then max-augment
		$s\to t$. Value $= F$ plus the extra max flow.

		\noindent\textbf{Min $s$--$t$ flow.}
		Binary search the capacity $C$ of $t\to s$; a
		feasible circulation exists iff a flow of value
		$\le C$ exists. Minimize $C$; the $s$--$t$ value
		is $C$ (flow on $t\to s$).
	\subsection{Kőnig, path cover, Dilworth}
		\noindent\textbf{Kőnig.}
		A vertex cover meets every edge. On a bipartite
		graph, max matching = min vertex cover; the
		complement of a min cover is a max independent
		set. Hall: a matching saturates $L$ iff
		$|N(S)|\ge|S|$ for every $S\subseteq L$.
		\texttt{cover} (Matching): from unmatched left
		vertices follow unmatched $L\to R$ and matched
		$R\to L$; the cover is unvisited left plus
		visited right. NP-hard on general graphs.

		\noindent\textbf{Path cover.}
		Min vertex-disjoint path cover of a DAG: split
		each $v$ into $v_L$ (left) and $v_R$ (right);
		each edge $u\to v$ becomes $u_L\to v_R$. Size
		$= n$ minus max matching. A match $u\mapsto v$
		means $u$ is followed by $v$ on a path.
		NP-hard on general digraphs.

		\noindent\textbf{Dilworth.}
		In a poset, min chain cover = max antichain.
		Build the comparability DAG ($u\to v$ iff $u<v$)
		and take a min path cover. Max antichain =
		$\{x:$ neither copy of $x$ is in the Kőnig
		cover$\}$.
		Mirsky (dual): min antichain cover = max chain;
		partition by longest-path height.
		On a sequence, min non-increasing-subsequence cover
		= strictly increasing LIS; min strictly decreasing
		subsequence cover = longest non-decreasing subsequence
		(LNDS). Equality may share a subsequence only in the
		first version: $[1,1]$ needs one non-increasing
		subsequence, but two strictly decreasing ones.

\section{Math}
	\subsection{Number of Spanning Trees}
		% I.e. matrix-tree theorem.
		% Source: https://en.wikipedia.org/wiki/Kirchhoff%27s_theorem
		% Test: stress-tests/graph/matrix-tree.cpp
		Create an $N\times N$ matrix \texttt{mat}, and for each edge $a \rightarrow b \in G$, do
		\texttt{mat[a][b]-{}-}, \texttt{mat[b][b]++} (and \texttt{mat[b][a]-{}-}, \texttt{mat[a][a]++} if $G$ is undirected).
		Delete row and column $i$ and take the determinant to get the number of directed spanning trees rooted at~$i$ (if undirected, any row/column).

	\subsection{Erdős–Gallai theorem}
		% Source: https://en.wikipedia.org/wiki/Erd%C5%91s%E2%80%93Gallai_theorem
		% Test: stress-tests/graph/erdos-gallai.cpp
		A simple graph with node degrees $d_1 \ge \dots \ge d_n$ exists iff $d_1 + \dots + d_n$ is even and for every $k = 1\dots n$,
		\[ \sum _{i=1}^{k}d_{i}\leq k(k-1)+\sum _{i=k+1}^{n}\min(d_{i},k). \]
